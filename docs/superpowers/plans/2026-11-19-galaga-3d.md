# Galaga 3D: Vanguard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a complete, playable 3D third-person Galaga-inspired rail-corridor arcade shooter ("Galaga 3D: Vanguard") in Godot 4.7 with procedural low-poly neon visuals, gamepad/keyboard dual-reticle aiming, 3D formation & diving choreography, Tractor Beam capture/rescue Dual Fighter mechanics, Challenge Stages, power-ups, multi-lock-on homing missiles, and the Stage 5 Queen Galaga Dreadnought boss.

**Architecture:** A cleanly decoupled Godot 4.7 scene-composition architecture centered around a typed `GameState` autoload signal bus and a `Main` stage orchestrator. All 3D meshes and arcade SFX are procedurally generated (`MeshFactory` and `SfxManager`) so the project has zero external asset dependencies, and every core gameplay system is verified both statically (`godot-plugin_godot:validate_scripts`) and behaviorally via standalone headless Godot scripts (`--headless --script`).

**Tech Stack:** Godot 4.7.2 (`/opt/homebrew/bin/godot`), strictly typed GDScript, Godot 3D Physics (`Area3D`/`CharacterBody3D`), `GPUParticles3D`, `WorldEnvironment` (ACES + Glow), `AudioStreamGenerator`, `godot-mcp` validation tools.

## Review Focus

- **Freed lock-on/homing target mid-flight:** An enemy painted by `ReticleRig` or tracked by an in-flight homing `Projectile` is destroyed by a primary laser or EMP before impact; both must check `is_instance_valid()` and drop stale locks or continue smoothly on last heading without errors (pinned in Task 2 & Task 3 tests).
- **Duplicate or illegal Tractor Beam trigger:** Two Boss Galagas attempting tractor dives concurrently, or triggering while player is `DUAL`, a captured ship exists, or within the `12.0s` cooldown; `FormationManager` must enforce all four preconditions atomically (pinned in Task 5 tests).
- **Captured fighter rescue vs. hostile turn vs. friendly fire:** Destroying a Boss while `DIVING` rescues and upgrades to `DUAL` (`+1000`); destroying it while `IN_FORMATION` turns the captured ship into a hostile diver (no upgrade); shooting the captured ship directly destroys it cleanly without crashing the Boss (pinned in Task 4 tests).
- **Atomic damage priority under simultaneous hits:** Overlapping enemy bullets/collisions in a single physics tick must follow `invuln_timer > 0` ignore → `shield_count` absorb → `DUAL` downgrade to `SINGLE` (`1.8s` invuln, no life lost) → life decrement, never stripping both shield and dual/life in one frame (pinned in Task 3 tests).
- **Challenge Stage completion with escaped enemies:** If enemies exit past `Z = +12` without being destroyed, the stage must still finish once `destroyed + escaped == 40`, granting `+1000` only for 8/8 waves and `+10,000` perfect bonus only when `destroyed == 40` (pinned in Task 5 tests).

```gdscript
class_name TestAssert
extends RefCounted

var failures: Array[String] = []
var checks: int = 0


func eq(actual: Variant, expected: Variant, label: String) -> void:
	checks += 1
	if actual != expected:
		failures.append("%s: expected %s, got %s" % [label, str(expected), str(actual)])


func is_true(cond: bool, label: String) -> void:
	checks += 1
	if not cond:
		failures.append("%s: expected true" % label)


func approx(actual: float, expected: float, eps: float, label: String) -> void:
	checks += 1
	if absf(actual - expected) > eps:
		failures.append("%s: expected ~%.4f, got %.4f" % [label, expected, actual])


func finish(tree: SceneTree, suite_name: String) -> void:
	if failures.is_empty():
		print("PASS [%s]: %d checks" % [suite_name, checks])
		tree.quit(0)
	else:
		printerr("FAIL [%s]: %d/%d failed" % [suite_name, failures.size(), checks])
		for f: String in failures:
			printerr("  - " + f)
		tree.quit(1)
```

```gdscript
	if stage_num % 5 == 0:
		return StageType.BOSS
	if stage_num % 3 == 0:
		return StageType.CHALLENGE
	return StageType.STANDARD


func add_score(points: int) -> void:
	if points <= 0:
		return
	score += points
	if score > high_score:
		high_score = score
	while score >= _next_extra_life:
		lives += 1
		_next_extra_life += NEXT_EXTRA_LIFE_STEP
		lives_changed.emit(lives)
	score_changed.emit(score, high_score)


func advance_stage() -> StageType:
	stage += 1
	var st: StageType = get_stage_type(stage)
	stage_changed.emit(stage, st)
	return st


func set_fighter_mode(mode: FighterMode) -> void:
	fighter_mode = mode
	var cap: int = 2 if fighter_mode == FighterMode.DUAL else 1
	if shield_count > cap:
		shield_count = cap
		shield_changed.emit(shield_count)
	fighter_mode_changed.emit(fighter_mode)


func grant_powerup(kind: PowerupType) -> void:
	match kind:
		PowerupType.SHIELD:
			var cap: int = 2 if fighter_mode == FighterMode.DUAL else 1
			shield_count = mini(shield_count + 1, cap)
			shield_changed.emit(shield_count)
		PowerupType.RAPID:
			rapid_timer = RAPID_DURATION
			rapid_changed.emit(rapid_timer)
		PowerupType.EMP:
			emp_charges = mini(emp_charges + 1, MAX_EMP_CHARGES)
			emp_changed.emit(emp_charges)


func consume_shield() -> bool:
	if shield_count <= 0:
		return false
	shield_count -= 1
	shield_changed.emit(shield_count)
	return true


func spend_emp() -> bool:
	if emp_charges <= 0:
		return false
	emp_charges -= 1
	emp_changed.emit(emp_charges)
	emp_triggered.emit()
	return true


func spend_lockon_energy(amount: float) -> bool:
	if lockon_energy + 0.001 < amount:
		return false
	lockon_energy = maxf(0.0, lockon_energy - amount)
	energy_changed.emit(lockon_energy)
	return true


func add_lockon_energy(amount: float) -> void:
	if amount <= 0.0:
		return
	var next_val: float = minf(MAX_ENERGY, lockon_energy + amount)
	if not is_equal_approx(next_val, lockon_energy):
		lockon_energy = next_val
		energy_changed.emit(lockon_energy)


func lose_life() -> bool:
	lives = maxi(0, lives - 1)
	lives_changed.emit(lives)
	if lives <= 0:
		is_game_over = true
		game_over_triggered.emit()
		return false
	return true


func tick_timers(delta: float) -> void:
	if rapid_timer > 0.0:
		rapid_timer = maxf(0.0, rapid_timer - delta)
		rapid_changed.emit(rapid_timer)
	add_lockon_energy(ENERGY_REGEN_PER_SEC * delta)

	return minf(1.8, 1.0 + 0.06 * float(maxi(0, stage_num - 1)))
```

```gdscript
func _make_sweep(f0: float, f1: float, duration: float, wave: String, gain: float) -> AudioStreamWAV:
	var count: int = int(SAMPLE_RATE * duration)
	var data := PackedByteArray()
	data.resize(count * 2)
	var phase: float = 0.0
	var rng := RandomNumberGenerator.new()
	rng.seed = int(f0 * 17.0 + duration * 1000.0)
	for i: int in range(count):
		var u: float = float(i) / float(maxi(1, count - 1))
		var freq: float = lerpf(f0, f1, u)
		if wave == "wobble":
			freq += sin(u * TAU * 7.0) * 90.0
		phase += TAU * freq / float(SAMPLE_RATE)
		var env: float = (1.0 - u) * minf(1.0, u * 25.0)
		var s: float = 0.0
		match wave:
			"square":
				s = 1.0 if sin(phase) >= 0.0 else -1.0
			"noise":
				s = rng.randf_range(-1.0, 1.0)
			"noise_mix":
				s = 0.55 * sin(phase) + 0.45 * rng.randf_range(-1.0, 1.0)
			_:
				s = sin(phase)
		var sample_i16: int = clampi(int(s * env * gain * 32767.0), -32768, 32767)
		data.encode_s16(i * 2, sample_i16)
	var wav := AudioStreamWAV.new()
	wav.format = AudioStreamWAV.FORMAT_16_BITS
	wav.mix_rate = SAMPLE_RATE
	wav.stereo = false
	wav.data = data
	return wav


func _make_arpeggio(freqs: Array, step_dur: float, gain: float) -> AudioStreamWAV:
	var step_samples: int = int(SAMPLE_RATE * step_dur)
	var total: int = step_samples * freqs.size()
	var data := PackedByteArray()
	data.resize(total * 2)
	var phase: float = 0.0
	for n: int in range(freqs.size()):
		var freq: float = float(freqs[n])
		for i: int in range(step_samples):
			var u: float = float(i) / float(maxi(1, step_samples - 1))
			phase += TAU * freq / float(SAMPLE_RATE)
			var env: float = (1.0 - u * 0.7) * minf(1.0, u * 20.0)
			var s: float = 0.65 * sin(phase) + 0.35 * (1.0 if sin(phase * 2.0) >= 0.0 else -1.0)
			var idx: int = (n * step_samples + i) * 2
			data.encode_s16(idx, clampi(int(s * env * gain * 32767.0), -32768, 32767))
	var wav := AudioStreamWAV.new()
	wav.format = AudioStreamWAV.FORMAT_16_BITS
	wav.mix_rate = SAMPLE_RATE
	wav.stereo = false
	wav.data = data
	return wav
```

```gdscript
class_name Projectile
extends Area3D

enum Kind { PLAYER_LASER, HOMING_MISSILE, ENEMY_ORB }

const TURN_RATE := 7.5
const MAX_LIFETIME := 4.0

var kind: Kind = Kind.PLAYER_LASER

var damage: int = 1
var homing_target: Node3D = null
var _speed: float = 120.0
var _age: float = 0.0
var _visual: Node3D = null


func _ready() -> void:
	area_entered.connect(_on_area_entered)
	body_entered.connect(_on_body_entered)


func setup(p_kind: Kind, origin: Vector3, dir: Vector3, speed: float, p_damage: int = 1, p_target: Node3D = null) -> void:
	kind = p_kind
	_speed = speed
	damage = p_damage
	homing_target = p_target
	global_position = origin
	var n_dir: Vector3 = dir.normalized() if dir.length_squared() > 0.0001 else Vector3(0.0, 0.0, -1.0)
	velocity = n_dir * _speed
	_configure_layers()
	_rebuild_visual()


func _configure_layers() -> void:
	if kind == Kind.ENEMY_ORB:
		collision_layer = 1 << 3 # Layer 4: enemy_projectile
		collision_mask = 1 << 0  # Layer 1: player
		add_to_group("enemy_projectiles")
	else:
		collision_layer = 1 << 1 # Layer 2: player_projectile
		collision_mask = 1 << 2  # Layer 3: enemy
		add_to_group("player_projectiles")


func _rebuild_visual() -> void:
	if is_instance_valid(_visual):
		_visual.queue_free()
	_visual = MeshFactory.build_projectile_mesh(kind)
	add_child(_visual)


func _physics_process(delta: float) -> void:
	step_motion(delta)


func step_motion(delta: float) -> void:
	_age += delta
	if _age >= MAX_LIFETIME or global_position.z < -175.0 or global_position.z > 22.0:
		queue_free()
		return
	if kind == Kind.HOMING_MISSILE:
		if homing_target != null and not is_instance_valid(homing_target):
			homing_target = null
		if homing_target != null:
			var desired: Vector3 = (homing_target.global_position - global_position).normalized()
			var cur_dir: Vector3 = velocity.normalized()
			var next_dir: Vector3 = cur_dir.lerp(desired, clampf(TURN_RATE * delta, 0.0, 1.0)).normalized()
			velocity = next_dir * _speed
	global_position += velocity * delta
	if velocity.length_squared() > 0.01 and is_inside_tree():
		var dir_n: Vector3 = velocity.normalized()
		if absf(dir_n.dot(Vector3.UP)) < 0.99:
			look_at(global_position + dir_n, Vector3.UP)


func _on_area_entered(area: Area3D) -> void:
	if kind != Kind.ENEMY_ORB and area.has_method("take_damage"):
		area.call("take_damage", damage, kind == Kind.HOMING_MISSILE)
		if kind == Kind.PLAYER_LASER:
			GameState.add_lockon_energy(6.0)
		queue_free()


func _on_body_entered(body: Node3D) -> void:
	if kind == Kind.ENEMY_ORB and body.has_method("apply_damage"):
		body.call("apply_damage", global_position.x)
		queue_free()
```

```gdscript
class_name ReticleRig
extends Node3D

const NEAR_Z := -22.0
const FAR_Z := -50.0
const AIM_MAGNET_RADIUS := 1.8
const LOCKON_CONE_RADIUS := 4.5
const MAX_LEAD_RAD := deg_to_rad(4.0)

var locked_targets: Array[Node3D] = []
var _near_mesh: MeshInstance3D
var _far_mesh: MeshInstance3D
var _aim_dir: Vector3 = Vector3(0.0, 0.0, -1.0)
var _ship_origin: Vector3 = Vector3.ZERO


func _ready() -> void:
	_near_mesh = _make_ring(0.85, 0.98, Color(0.2, 1.0, 0.9))
	_far_mesh = _make_ring(0.45, 0.56, Color(1.0, 0.9, 0.25))
	add_child(_near_mesh)
	add_child(_far_mesh)


func _make_ring(inner_r: float, outer_r: float, col: Color) -> MeshInstance3D:
	var mi := MeshInstance3D.new()
	var tm := TorusMesh.new()
	tm.inner_radius = inner_r
	tm.outer_radius = outer_r
	mi.mesh = tm
	mi.rotation_degrees = Vector3(90.0, 0.0, 0.0)
	mi.material_override = MeshFactory.make_neon_mat(col, col, 2.8, 0.85)
	return mi


func update_aim(ship_pos: Vector3, move_vec: Vector2, fine_aim: Vector2) -> void:
	_ship_origin = ship_pos
	var yaw: float = clampf(-(move_vec.x * 0.55 + fine_aim.x) * MAX_LEAD_RAD, -MAX_LEAD_RAD, MAX_LEAD_RAD)
	var pitch: float = clampf((move_vec.y * 0.55 - fine_aim.y) * MAX_LEAD_RAD, -MAX_LEAD_RAD, MAX_LEAD_RAD)
	_aim_dir = Vector3(sin(yaw), sin(pitch), -1.0).normalized()
	if is_instance_valid(_near_mesh):
		_near_mesh.global_position = _project_at_z(NEAR_Z)
	if is_instance_valid(_far_mesh):
		_far_mesh.global_position = _project_at_z(FAR_Z)


func _project_at_z(target_z: float) -> Vector3:
	var t: float = (target_z - _ship_origin.z) / minf(-0.001, _aim_dir.z)
	return _ship_origin + _aim_dir * t


func get_far_reticle_world() -> Vector3:
	return _project_at_z(FAR_Z)


func compute_assisted_dir(muzzle_world: Vector3, candidates: Array) -> Vector3:
	var best_pos: Vector3 = get_far_reticle_world()
	var best_dist: float = AIM_MAGNET_RADIUS
	for c: Variant in candidates:
		if not is_instance_valid(c) or not (c is Node3D):
			continue
		var node: Node3D = c as Node3D
		if node.global_position.z >= muzzle_world.z - 2.0:
			continue
		var ray_pt: Vector3 = _project_at_z(node.global_position.z)
		var lateral: float = Vector2(node.global_position.x - ray_pt.x, node.global_position.y - ray_pt.y).length()
		if lateral < best_dist:
			best_dist = lateral
			best_pos = node.global_position
	return (best_pos - muzzle_world).normalized()


func prune_locked_targets() -> void:
	var alive: Array[Node3D] = []
	for t_node: Node3D in locked_targets:
		if is_instance_valid(t_node):
			alive.append(t_node)
	locked_targets = alive


func scan_lockons(candidates: Array, max_locks: int = 4) -> Array[Node3D]:
	prune_locked_targets()
	if locked_targets.size() >= max_locks:
		return locked_targets
	var scored: Array[Dictionary] = []
	for c: Variant in candidates:
		if not is_instance_valid(c) or not (c is Node3D):
			continue
		var node: Node3D = c as Node3D
		if locked_targets.has(node):
			continue
		if node.global_position.z >= -2.0 or node.global_position.z < -120.0:
			continue
		var ray_pt: Vector3 = _project_at_z(node.global_position.z)
		var lateral: float = Vector2(node.global_position.x - ray_pt.x, node.global_position.y - ray_pt.y).length()
		if lateral <= LOCKON_CONE_RADIUS:
			var priority: float = lateral
			if node.has_method("is_high_priority_lock") and node.call("is_high_priority_lock"):
				priority -= 2.5
			scored.append({"node": node, "score": priority})
	scored.sort_custom(func(a: Dictionary, b: Dictionary) -> bool: return float(a["score"]) < float(b["score"]))
	for item: Dictionary in scored:
		if locked_targets.size() >= max_locks:
			break
		locked_targets.append(item["node"] as Node3D)
		SfxManager.play("lockon")
	return locked_targets


func clear_lockons() -> void:
	locked_targets.clear()
```

```gdscript
class_name TractorBeam
extends Area3D

signal capture_completed(player: PlayerFighter)

const CAPTURE_DWELL := 0.65
const BEAM_RADIUS_AT_PLAYER := 2.8

var _dwell: float = 0.0
var _rings: Array[MeshInstance3D] = []
var _time: float = 0.0


func _ready() -> void:
	collision_layer = 1 << 4 # Layer 5: tractor_beam
	collision_mask = 1 << 0  # Layer 1: player
	_build_visual()


func _build_visual() -> void:
	var cone := MeshInstance3D.new()
	var cm := CylinderMesh.new()
	cm.top_radius = 0.45
	cm.bottom_radius = BEAM_RADIUS_AT_PLAYER
	cm.height = 15.0
	cone.mesh = cm
	cone.rotation_degrees = Vector3(90.0, 0.0, 0.0)
	cone.position = Vector3(0.0, 0.0, 7.5)
	cone.material_override = MeshFactory.make_neon_mat(Color(0.2, 0.9, 1.0), Color(0.55, 0.3, 1.0), 2.4, 0.28)
	add_child(cone)
	for i: int in range(4):
		var mi := MeshInstance3D.new()
		var tm := TorusMesh.new()
		tm.inner_radius = 0.7 + float(i) * 0.45
		tm.outer_radius = tm.inner_radius + 0.14
		mi.mesh = tm
		mi.rotation_degrees = Vector3(90.0, 0.0, 0.0)
		mi.position = Vector3(0.0, 0.0, 3.0 + float(i) * 3.0)
		mi.material_override = MeshFactory.make_neon_mat(Color(0.3, 1.0, 1.0), Color(0.65, 0.35, 1.0), 3.0, 0.75)
		add_child(mi)
		_rings.append(mi)


func _process(delta: float) -> void:
	_time += delta
	for i: int in range(_rings.size()):
		var phase: float = fmod(_time * 1.4 + float(i) * 0.25, 1.0)
		_rings[i].position.z = lerpf(1.5, 14.5, phase)
		var s: float = lerpf(0.35, 1.35, phase)
		_rings[i].scale = Vector3(s, s, 1.0)


func tick_beam(delta: float, player: PlayerFighter) -> bool:
	if not is_instance_valid(player) or player.is_captured or player.invuln_timer > 0.0:
		_dwell = maxf(0.0, _dwell - delta * 2.0)
		return false
	var lateral: float = Vector2(player.global_position.x - global_position.x, player.global_position.y - global_position.y).length()
	if lateral <= BEAM_RADIUS_AT_PLAYER and absf(player.global_position.z - (global_position.z + 14.0)) <= 4.0:
		_dwell += delta
		if _dwell >= CAPTURE_DWELL:
			_dwell = 0.0
			capture_completed.emit(player)
			return true
	else:
		_dwell = maxf(0.0, _dwell - delta * 1.5)
	return false
```

```gdscript
				global_position = slot_world
				state = State.FORMATION
		State.CAPTURED_HOSTILE:
			_t += delta * 0.55 * speed_mult
			global_position = _dive_start.lerp(Vector3(_dive_target.x, _dive_target.y, 14.0), _t)
			if _t >= 1.0:
				escaped.emit(self)
				queue_free()
				return
		State.RESCUED_DOCKING:
			if not is_instance_valid(_player):
				queue_free()
				return
			var dock_pos: Vector3 = _player.global_position + Vector3(1.25, 0.0, 0.0)
			global_position = global_position.lerp(dock_pos, clampf(5.5 * delta, 0.0, 1.0))
			rotation.z += delta * 6.0
			if global_position.distance_to(dock_pos) < 0.7:
				complete_rescue_dock()
				return
		State.CHALLENGE:
			_t += delta * 0.24 * speed_mult
			global_position = WavePatterns.sample_challenge_path(challenge_wave, _t)
			if _t >= 1.0:
				escaped.emit(self)
				queue_free()
				return

	if is_instance_valid(captured_child) and captured_child.state == State.CAPTURED_ATTACHED:
		captured_child.global_position = global_position + Vector3(0.0, 2.2, 0.0)


func _handle_player_captured() -> void:
	_despawn_beam()
	if not is_instance_valid(_player):
		state = State.RETURNING
		tractor_finished.emit(self)
		return
	_player.begin_capture(self)
	var cap: EnemyUnit = load("res://scenes/enemies/enemy_unit.tscn").instantiate()
	get_parent().add_child(cap)
	cap.setup_unit(Archetype.CAPTURED_FIGHTER, row, col, squad_index, global_position + Vector3(0.0, 2.2, 0.0), _player)
	attach_captured_fighter(cap)
	state = State.RETURNING
	capture_succeeded.emit(self, cap)
	tractor_finished.emit(self)


func complete_rescue_dock() -> void:
	if _dead:
		return
	_dead = true
	GameState.set_fighter_mode(GameState.FighterMode.DUAL)
	GameState.add_score(1000)
	SfxManager.play("dock")
	rescue_docked.emit(self)
	queue_free()


func _despawn_beam() -> void:
	if is_instance_valid(_beam):
		_beam.queue_free()
		_beam = null


func _fire_enemy_orb() -> void:
	if not is_instance_valid(_player):
		return
	var orb: Projectile = PROJECTILE_SCENE.instantiate()
	get_parent().add_child(orb)
	var dir: Vector3 = (_player.global_position - global_position).normalized()
	orb.setup(Projectile.Kind.ENEMY_ORB, global_position + Vector3(0.0, 0.0, 1.2), dir, 36.0 * speed_mult, 1)


func take_damage(amount: int, _from_missile: bool = false) -> void:
	if _dead or state == State.RESCUED_DOCKING:
		return
	hp -= amount
	if hp > 0:
		if archetype == Archetype.BOSS_GALAGA:
			_rebuild_visual(true)
		SfxManager.play("hit")
		return
	_dead = true
	_despawn_beam()
	var was_diving: bool = (state == State.DIVING or state == State.TRACTOR_BEAM)
	if is_instance_valid(captured_child) and captured_child.state == State.CAPTURED_ATTACHED:
		if was_diving:
			captured_child.state = State.RESCUED_DOCKING
			captured_child.remove_from_group("lockable_enemies")
			if is_instance_valid(captured_child._visual):
				captured_child._visual.queue_free()
			captured_child._visual = MeshFactory.build_player_ship(Color(0.25, 1.0, 0.9))
			captured_child.add_child(captured_child._visual)
		else:
			captured_child.state = State.CAPTURED_HOSTILE
			captured_child._t = 0.0
			captured_child._dive_start = captured_child.global_position
			captured_child._dive_target = _player.global_position if is_instance_valid(_player) else Vector3.ZERO
	captured_child = null

	var pts: int = _compute_score(was_diving)
	var should_drop: bool = false
	if was_diving and archetype == Archetype.BOSS_GALAGA:
		should_drop = true
	elif was_diving and archetype == Archetype.GOEI and randf() < 0.15:
		should_drop = true
	SfxManager.play("explode")
	destroyed.emit(self, pts, should_drop, global_position)
	queue_free()


func _compute_score(was_diving: bool) -> int:
	if state == State.CHALLENGE:
		return 100
	match archetype:
		Archetype.ZAKO:
			return 100 if was_diving else 50
		Archetype.GOEI:
			return 160 if was_diving else 80
		Archetype.BOSS_GALAGA:
			if not was_diving:
				return 150
			if escort_count_on_dive >= 2:
				return 1600
			if escort_count_on_dive == 1:
				return 800
			return 400
		_: # CAPTURED_FIGHTER shot directly
			return 1000 if state == State.CAPTURED_HOSTILE else 0


func _on_body_entered(body: Node3D) -> void:
	if _dead or state == State.CHALLENGE or state == State.RESCUED_DOCKING:
		return
	if body.has_method("apply_damage"):
		body.call("apply_damage", global_position.x)
		take_damage(99)
```

```gdscript
class_name FormationManager
extends Node3D

signal wave_cleared
signal challenge_completed(hits: int, total: int, wave_bonuses: int, perfect_bonus: int)

const ENEMY_SCENE: PackedScene = preload("res://scenes/enemies/enemy_unit.tscn")
const CAPSULE_SCENE: PackedScene = preload("res://scenes/pickups/powerup_capsule.tscn")
const TRACTOR_COOLDOWN_SEC := 12.0
const MAX_DROPS_PER_WAVE := 2

var tractor_cooldown: float = 8.0
var is_challenge: bool = false

var _player: PlayerFighter = null
var _stage_num: int = 1
var _sway_time: float = 0.0
var _spawn_queue: Array[Dictionary] = []
var _spawn_timer: float = 0.0
var _dive_timer: float = 2.0
var _units: Array[EnemyUnit] = []
var _active_tractor_boss: EnemyUnit = null
var _captured_unit: EnemyUnit = null
var _drops_this_wave: int = 0
var _drop_cycle: int = 0
var _finished: bool = false

# Challenge tracking
var _challenge_hits: int = 0
var _challenge_resolved: int = 0
var _challenge_wave_hits: Array[int] = [0, 0, 0, 0, 0]


func _ready() -> void:
	GameState.emp_triggered.connect(handle_emp_pulse)


func start_standard_wave(stage_num: int, player: PlayerFighter) -> void:
	_reset_internal(stage_num, player, false)
	# Build 40-enemy queue grouped into 5 squads of 8
	var all_slots: Array[Dictionary] = []
	for r: int in range(WavePatterns.ROW_COUNTS.size()):
		var arch: EnemyUnit.Archetype = EnemyUnit.Archetype.BOSS_GALAGA if r == 0 else (EnemyUnit.Archetype.GOEI if r <= 2 else EnemyUnit.Archetype.ZAKO)
		for c: int in range(WavePatterns.ROW_COUNTS[r]):
			all_slots.append({"arch": arch, "row": r, "col": c})
	for i: int in range(all_slots.size()):
		var sq: int = int(i / 8)
		var entry: Dictionary = all_slots[i]
		entry["squad"] = sq
		entry["delay"] = float(sq) * 1.45 + float(i % 8) * 0.16
		_spawn_queue.append(entry)


func start_challenge_stage(stage_num: int, player: PlayerFighter) -> void:
	_reset_internal(stage_num, player, true)
	for w: int in range(5):
		for k: int in range(8):
			var arch: EnemyUnit.Archetype = EnemyUnit.Archetype.GOEI if w % 2 == 1 else EnemyUnit.Archetype.ZAKO
			_spawn_queue.append({
				"arch": arch,
				"row": 0,
				"col": k,
				"squad": w,
				"challenge_wave": w,
				"delay": float(w) * 3.4 + float(k) * 0.22,
			})


func _reset_internal(stage_num: int, player: PlayerFighter, challenge_flag: bool) -> void:
	_stage_num = stage_num
	_player = player
	is_challenge = challenge_flag
	_sway_time = 0.0
	_spawn_timer = 0.0
	_dive_timer = 2.4
	_spawn_queue.clear()
	_units.clear()
	_active_tractor_boss = null
	_captured_unit = null
	_drops_this_wave = 0
	_finished = false
	_challenge_hits = 0
	_challenge_resolved = 0
	_challenge_wave_hits = [0, 0, 0, 0, 0]
	tractor_cooldown = 8.0


func _physics_process(delta: float) -> void:
	if _finished or GameState.is_game_over:
		return
	_sway_time += delta
	_spawn_timer += delta
	if tractor_cooldown > 0.0:
		tractor_cooldown = maxf(0.0, tractor_cooldown - delta)

	# Process spawn queue
	while not _spawn_queue.is_empty() and float(_spawn_queue[0]["delay"]) <= _spawn_timer:
		var item: Dictionary = _spawn_queue.pop_front()
		_spawn_unit(item)

	# Update formation slot sway
	var sway_offset := Vector3(sin(_sway_time * 1.1) * 2.0, sin(_sway_time * 1.7) * 0.6, 0.0)
	_prune_units()
	for u: EnemyUnit in _units:
		if not is_challenge:
			u.slot_world = WavePatterns.get_formation_slot_local(u.row, u.col) + sway_offset

	if not is_challenge:
		_dive_timer -= delta
		if _dive_timer <= 0.0 and _spawn_queue.is_empty():
			_schedule_next_dive()
			_dive_timer = maxf(1.2, 2.8 - float(_stage_num - 1) * 0.14)
		if _spawn_queue.is_empty() and _units.is_empty():
			_finished = true
			wave_cleared.emit()


func _spawn_unit(item: Dictionary) -> void:
	var u: EnemyUnit = ENEMY_SCENE.instantiate()
	add_child(u)
	var slot: Vector3 = WavePatterns.get_formation_slot_local(int(item["row"]), int(item["col"]))
	u.setup_unit(
		int(item["arch"]) as EnemyUnit.Archetype,
		int(item["row"]),
		int(item["col"]),
		int(item["squad"]),
		slot,
		_player,
		is_challenge,
		int(item.get("challenge_wave", 0))
	)
	u.destroyed.connect(_on_unit_destroyed)
	u.escaped.connect(_on_unit_escaped)
	u.capture_succeeded.connect(_on_capture_succeeded)
	u.tractor_finished.connect(_on_tractor_finished)
	_units.append(u)


func _prune_units() -> void:
	var alive: Array[EnemyUnit] = []
	for u: EnemyUnit in _units:
		if is_instance_valid(u):
			alive.append(u)
	_units = alive
	if _active_tractor_boss != null and not is_instance_valid(_active_tractor_boss):
		_active_tractor_boss = null
	if _captured_unit != null and not is_instance_valid(_captured_unit):
		_captured_unit = null


func can_start_tractor_beam() -> bool:
	_prune_units()
	return (
		not is_challenge
		and GameState.fighter_mode == GameState.FighterMode.SINGLE
		and _captured_unit == null
		and _active_tractor_boss == null
		and tractor_cooldown <= 0.0
	)


func try_trigger_tractor_beam(boss: EnemyUnit) -> bool:
	if not can_start_tractor_beam():
		return false
	if not is_instance_valid(boss) or boss.archetype != EnemyUnit.Archetype.BOSS_GALAGA or boss.state != EnemyUnit.State.FORMATION:
		return false
	_active_tractor_boss = boss
	tractor_cooldown = TRACTOR_COOLDOWN_SEC
	boss.start_tractor_dive()
	return true


func _schedule_next_dive() -> void:
	var diving_count: int = 0
	var ready_bosses: Array[EnemyUnit] = []
	var ready_others: Array[EnemyUnit] = []
	for u: EnemyUnit in _units:
		if u.state == EnemyUnit.State.DIVING or u.state == EnemyUnit.State.TRACTOR_BEAM:
			diving_count += 1
		elif u.state == EnemyUnit.State.FORMATION:
			if u.archetype == EnemyUnit.Archetype.BOSS_GALAGA:
				ready_bosses.append(u)
			else:
				ready_others.append(u)
	if diving_count >= GameState.get_max_divers(_stage_num):
		return

	# Prefer tractor beam if eligible
	if can_start_tractor_beam() and not ready_bosses.is_empty():
		var chosen_boss: EnemyUnit = ready_bosses[randi() % ready_bosses.size()]
		if try_trigger_tractor_beam(chosen_boss):
			return

	# Boss + up to 2 Goei escorts, or single diver
	if not ready_bosses.is_empty() and randf() < 0.35:
		var boss: EnemyUnit = ready_bosses[randi() % ready_bosses.size()]
		var escorts: Array[EnemyUnit] = []
		for u: EnemyUnit in ready_others:
			if u.archetype == EnemyUnit.Archetype.GOEI and escorts.size() < 2:
				escorts.append(u)
		boss.start_dive(escorts.size())
		for e: EnemyUnit in escorts:
			e.start_dive(0)
	elif not ready_others.is_empty():
		var diver: EnemyUnit = ready_others[randi() % ready_others.size()]
		diver.start_dive(0)


func _on_unit_destroyed(unit: EnemyUnit, score_award: int, drop_capsule: bool, world_pos: Vector3) -> void:
	GameState.add_score(score_award)
	_units.erase(unit)
	if unit == _active_tractor_boss:
		_active_tractor_boss = null
	if unit == _captured_unit:
		_captured_unit = null
	if is_challenge:
		report_challenge_outcome(unit.challenge_wave, true)
		return
	if drop_capsule and _drops_this_wave < MAX_DROPS_PER_WAVE:
		_drops_this_wave += 1
		var cap: PowerupCapsule = CAPSULE_SCENE.instantiate()
		get_parent().add_child(cap)
		var kind: GameState.PowerupType = (_drop_cycle % 3) as GameState.PowerupType
		_drop_cycle += 1
		cap.setup(kind, world_pos, _player)


func _on_unit_escaped(unit: EnemyUnit) -> void:
	_units.erase(unit)
	if is_challenge:
		report_challenge_outcome(unit.challenge_wave, false)


func _on_capture_succeeded(_boss: EnemyUnit, captured_unit: EnemyUnit) -> void:
	_captured_unit = captured_unit
	captured_unit.destroyed.connect(_on_unit_destroyed)
	captured_unit.escaped.connect(_on_unit_escaped)
	captured_unit.rescue_docked.connect(func(_u: EnemyUnit) -> void: _captured_unit = null)


func _on_tractor_finished(boss: EnemyUnit) -> void:
	if boss == _active_tractor_boss:
		_active_tractor_boss = null


func report_challenge_outcome(wave_idx: int, was_destroyed: bool) -> void:
	if _finished or not is_challenge:
		return
	_challenge_resolved += 1
	if was_destroyed:
		_challenge_hits += 1
		if wave_idx >= 0 and wave_idx < _challenge_wave_hits.size():
			_challenge_wave_hits[wave_idx] += 1
	if _challenge_resolved >= 40:
		_finished = true
		var wave_bonuses: int = 0
		for hits_in_wave: int in _challenge_wave_hits:
			if hits_in_wave >= 8:
				wave_bonuses += 1000
```

```gdscript
extends Node3D

	_player = player
	phase = 1
	left_hp = 18 + stage_num * 2
	right_hp = 18 + stage_num * 2

		c.queue_free()

func _physics_process(delta: float) -> void:
	if _defeated or GameState.is_game_over:
		return
	if _entering:
		global_position.z = lerpf(global_position.z, -55.0, clampf(2.2 * delta, 0.0, 1.0))
		if absf(global_position.z - (-55.0)) < 0.8:
			global_position.z = -55.0
			_entering = false
		return

	_sway_t += delta * (1.25 if phase == 2 else 0.75)
	var lunge_z: float = sin(_sway_t * 1.4) * 8.0 if phase == 2 else 0.0
	global_position = Vector3(sin(_sway_t) * 6.0, sin(_sway_t * 2.0) * 2.0 + 1.5, -55.0 + lunge_z)

	if left_hp > 0:
		_cannon_timer -= delta
		if _cannon_timer <= 0.0:
			_cannon_timer = 2.6
			_fire_cannon_barrage()

	if right_hp > 0:
		_hangar_timer -= delta
		if _hangar_timer <= 0.0:
			_hangar_timer = 8.0
			_launch_drone_squad()

	if phase == 2:
		_spiral_timer -= delta
		if _spiral_timer <= 0.0:
			_spiral_timer = 0.85
			_fire_phase2_spiral()


func _fire_cannon_barrage() -> void:
	if not is_instance_valid(_player) or not is_instance_valid(_left_area):
		return
	var origin: Vector3 = _left_area.global_position + Vector3(0.0, 0.0, 3.0)
	for spread: float in [-0.14, 0.0, 0.14]:
		var dir: Vector3 = (_player.global_position - origin).normalized()
		dir = dir.rotated(Vector3.UP, spread)
		var orb: Projectile = PROJECTILE_SCENE.instantiate()
		get_parent().add_child(orb)
		orb.setup(Projectile.Kind.ENEMY_ORB, origin, dir, 40.0, 1)

		return
	for i: int in range(3):
		var u: EnemyUnit = ENEMY_SCENE.instantiate()
		get_parent().add_child(u)
		var spawn_pos: Vector3 = _right_area.global_position + Vector3(float(i - 1) * 1.5, 0.0, 3.0)
		u.setup_unit(EnemyUnit.Archetype.ZAKO, 3, i, 0, spawn_pos, _player)
		u.global_position = spawn_pos
		u.state = EnemyUnit.State.FORMATION
		u.start_dive(0)
		u.destroyed.connect(func(_unit: EnemyUnit, pts: int, _drop: bool, _pos: Vector3) -> void: GameState.add_score(pts))

	if not is_instance_valid(_player):
		return
	var origin: Vector3 = global_position + Vector3(0.0, 0.0, 3.2)
	var base_dir: Vector3 = (_player.global_position - origin).normalized()
	_spiral_angle += 0.45
	for i: int in range(6):
		var ang: float = _spiral_angle + float(i) * (TAU / 6.0)
		var offset := Vector3(cos(ang) * 0.24, sin(ang) * 0.24, 0.0)
		var orb: Projectile = PROJECTILE_SCENE.instantiate()
		get_parent().add_child(orb)
		orb.setup(Projectile.Kind.ENEMY_ORB, origin, (base_dir + offset).normalized(), 44.0, 1)


func apply_part_damage(part_name: String, raw_damage: int) -> int:
	if _defeated or raw_damage <= 0:
		return 0
	var actual: int = raw_damage
	match part_name:
		"cannon":
			if left_hp <= 0:
				return 0
			left_hp = maxi(0, left_hp - raw_damage)
			if left_hp == 0 and is_instance_valid(_left_area):
				_left_area.queue_free()
				SfxManager.play("explode")
		"hangar":
			if right_hp <= 0:
				return 0
			right_hp = maxi(0, right_hp - raw_damage)
			if right_hp == 0 and is_instance_valid(_right_area):
				_right_area.queue_free()
				SfxManager.play("explode")
		"core":
			if phase == 1:
				actual = maxi(1, int(floor(float(raw_damage) * 0.5)))
			core_hp = maxi(0, core_hp - actual)
	if phase == 1 and left_hp <= 0 and right_hp <= 0:
		phase = 2
		SfxManager.play("emp")
	SfxManager.play("hit")
	boss_hp_changed.emit(get_total_hp_ratio(), phase)
	if core_hp <= 0:
		_defeated = true
		var award: int = 5000 * maxi(1, int(_stage_num / 5))
		GameState.add_score(award)
		GameState.grant_powerup(GameState.PowerupType.SHIELD)
		SfxManager.play("boss_explode")
		boss_defeated.emit(award)
		queue_free()
	return actual


func get_total_hp_ratio() -> float:
	return float(maxi(0, core_hp + left_hp + right_hp)) / float(maxi(1, _max_total_hp))
```

```gdscript
	_boss_bar.modulate = Color(1.0, 0.25, 0.25) if phase == 2 else Color(0.25, 0.85, 1.0)


func update_lockon_brackets(camera: Camera3D, targets: Array[Node3D]) -> void:
	for c: Node in _bracket_layer.get_children():
		c.queue_free()
	if not is_instance_valid(camera):
		return
	for target: Node3D in targets:
		if not is_instance_valid(target) or camera.is_position_behind(target.global_position):
			continue
		var screen_pos: Vector2 = camera.unproject_position(target.global_position)
		var lbl := _make_label("[ + ]", 20, Color(1.0, 0.55, 0.1))
		lbl.position = screen_pos - Vector2(22, 14)
		_bracket_layer.add_child(lbl)
```

```gdscript
extends Node3D

const PLAYER_SCENE: PackedScene = preload("res://scenes/player/player_fighter.tscn")
const FORMATION_SCENE: PackedScene = preload("res://scenes/enemies/formation_manager.tscn")

var _active_encounter: Node3D = null

func _ready() -> void:
	_camera = $Camera3D
	_hud = $HUD
	_world_root = $WorldRoot
	_build_environment_and_tunnel()
	_hud.start_requested.connect(_start_new_run)

func _build_environment_and_tunnel() -> void:
	# Streaming 3D starfield GPUParticles3D
	var stars := GPUParticles3D.new()
	stars.amount = 360
	stars.lifetime = 3.2
	stars.preprocess = 3.2
	stars.position = Vector3(0.0, 0.0, -175.0)
	var pm := ParticleProcessMaterial.new()
	pm.emission_shape = ParticleProcessMaterial.EMISSION_SHAPE_BOX
	pm.emission_box_extents = Vector3(45.0, 26.0, 4.0)
	pm.direction = Vector3(0.0, 0.0, 1.0)
	pm.spread = 0.0
	pm.initial_velocity_min = 52.0
	pm.initial_velocity_max = 88.0
	pm.gravity = Vector3.ZERO
	stars.process_material = pm
	var star_mesh := BoxMesh.new()
	star_mesh.size = Vector3(0.14, 0.14, 1.8)
	star_mesh.material = MeshFactory.make_neon_mat(Color(0.85, 0.95, 1.0), Color(0.7, 0.9, 1.0), 2.4)
	stars.draw_pass_1 = star_mesh
	add_child(stars)

	# Neon corridor depth rings for speed reference
	var ring_mat := MeshFactory.make_neon_mat(Color(0.15, 0.45, 0.95), Color(0.15, 0.5, 1.0), 1.6, 0.22)
	for i: int in range(8):

		var tm := TorusMesh.new()
		tm.inner_radius = 22.0
		tm.outer_radius = 22.28
		mi.mesh = tm
		mi.rotation_degrees = Vector3(90.0, 0.0, 0.0)
		mi.position = Vector3(0.0, 0.0, -20.0 - float(i) * 22.0)
		mi.material_override = ring_mat
		add_child(mi)
		_depth_rings.append(mi)


func _start_new_run() -> void:
	get_tree().paused = false
	_clear_world()
	GameState.reset_run()
	_hud.show_title(false)
	_hud.show_pause(false)
	_hud.set_boss_bar(false)

	_player.barrel_roll_started.connect(func() -> void: _fov_kick = 5.0)
	_player.captured_by_boss.connect(_on_player_captured)
	_in_run = true
	_transitioning = false
	_launch_stage(GameState.stage)


func _clear_world() -> void:
	for c: Node in _world_root.get_children():
		c.queue_free()
	_player = null
	_active_encounter = null


func _launch_stage(stage_num: int) -> void:
	for proj: Node in get_tree().get_nodes_in_group("enemy_projectiles"):
		if is_instance_valid(proj):
			proj.queue_free()
	if is_instance_valid(_active_encounter):
		_active_encounter.queue_free()
		_active_encounter = null
	var st: GameState.StageType = GameState.get_stage_type(stage_num)
	match st:
		GameState.StageType.BOSS:
			_hud.show_banner("WARNING: STAGE %d" % stage_num, "QUEEN GALAGA DREADNOUGHT", 2.4)
			var boss: QueenBoss = BOSS_SCENE.instantiate()
			_world_root.add_child(boss)
			boss.setup_boss(stage_num, _player)
			boss.boss_hp_changed.connect(func(r: float, ph: int) -> void: _hud.set_boss_bar(true, r, ph))
			boss.boss_defeated.connect(_on_boss_defeated)
			_active_encounter = boss
		GameState.StageType.CHALLENGE:
			_hud.set_boss_bar(false)
			_hud.show_banner("CHALLENGE STAGE %d" % stage_num, "DESTROY ALL 40 DRONES FOR +10,000 PERFECT", 2.4)
			var fm: FormationManager = FORMATION_SCENE.instantiate()
			_world_root.add_child(fm)
			fm.start_challenge_stage(stage_num, _player)
			fm.challenge_completed.connect(_on_challenge_completed)
			_active_encounter = fm
		_:
			_hud.set_boss_bar(false)
			_hud.show_banner("STAGE %d" % stage_num, "INTERCEPT INCOMING FORMATION", 2.0)
			var fm: FormationManager = FORMATION_SCENE.instantiate()
			_world_root.add_child(fm)
			fm.start_standard_wave(stage_num, _player)
			fm.wave_cleared.connect(_on_wave_cleared)

func _on_wave_cleared() -> void:
	if _transitioning or GameState.is_game_over:
		return
	_transitioning = true
	SfxManager.play("stage_clear")
	_hud.show_banner("STAGE %d CLEAR" % GameState.stage, "PREPARING NEXT WAVE", 2.0)
	await get_tree().create_timer(2.2).timeout
	if not GameState.is_game_over and _in_run:
		GameState.advance_stage()
		_transitioning = false
		_launch_stage(GameState.stage)


func _on_challenge_completed(hits: int, total: int, wave_bonuses: int, perfect_bonus: int) -> void:
	if _transitioning or GameState.is_game_over:
		return
	_transitioning = true
	SfxManager.play("stage_clear")
	var sub: String = "HITS %d/%d  |  WAVE +%d  |  PERFECT +%d" % [hits, total, wave_bonuses, perfect_bonus]
	_hud.show_banner("CHALLENGE COMPLETE", sub, 3.0)
	await get_tree().create_timer(3.2).timeout
	if not GameState.is_game_over and _in_run:
		GameState.advance_stage()
		_transitioning = false
		_launch_stage(GameState.stage)


func _on_boss_defeated(award: int) -> void:
	if _transitioning or GameState.is_game_over:
		return
	_transitioning = true
	_shake_trauma = 1.0
	_hud.set_boss_bar(false)
	_hud.show_banner("DREADNOUGHT DESTROYED", "+%d  |  SHIELD RECHARGED" % award, 3.0)
	await get_tree().create_timer(3.2).timeout
	if not GameState.is_game_over and _in_run:
		GameState.advance_stage()
		_transitioning = false
		_launch_stage(GameState.stage)


func _on_player_captured(_boss: Node3D) -> void:
	_shake_trauma = 0.8
	_hud.show_banner("FIGHTER CAPTURED!", "DESTROY BOSS WHILE DIVING TO RESCUE", 2.2)
	if GameState.is_game_over:
		return
	await get_tree().create_timer(1.5).timeout
	if is_instance_valid(_player) and not GameState.is_game_over:
		_player.respawn_single()


func _on_game_over() -> void:
	_in_run = false
	_hud.set_boss_bar(false)
	_hud.show_game_over(GameState.score, GameState.high_score)


func _toggle_pause() -> void:
	if not _in_run or GameState.is_game_over:
		return
	var next_pause: bool = not get_tree().paused
	get_tree().paused = next_pause
	_hud.show_pause(next_pause)


func _process(delta: float) -> void:
	# Scroll neon corridor depth rings
	for ring: MeshInstance3D in _depth_rings:
		ring.position.z += 58.0 * delta
		if ring.position.z > 16.0:
			ring.position.z -= 176.0

	if not _in_run or get_tree().paused:
		return

	GameState.tick_timers(delta)
	_update_camera(delta)
	if is_instance_valid(_player):
		_hud.update_lockon_brackets(_camera, _player.get_locked_targets())


func _update_camera(delta: float) -> void:
	if not is_instance_valid(_player):
		return
	var p: Vector3 = _player.global_position
	var desired := Vector3(p.x * 0.25, p.y * 0.20 + 3.2, 12.5)
	_camera.global_position = _camera.global_position.lerp(desired, clampf(6.0 * delta, 0.0, 1.0))
	if _shake_trauma > 0.0:
		var amp: float = _shake_trauma * _shake_trauma * 0.65
		_camera.global_position += Vector3(randf_range(-amp, amp), randf_range(-amp, amp), 0.0)
		_shake_trauma = maxf(0.0, _shake_trauma - delta * 2.2)
	_fov_kick = lerpf(_fov_kick, 0.0, clampf(6.0 * delta, 0.0, 1.0))
	_camera.fov = 60.0 + _fov_kick
	_camera.look_at(Vector3(p.x * 0.15, p.y * 0.10, -35.0), Vector3.UP)
```
